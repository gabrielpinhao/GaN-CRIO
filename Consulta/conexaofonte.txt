import pyvisa
import time

rm = pyvisa.ResourceManager()
print(rm.list_resources())

try:
    instrumento = rm.open_resource('GPIB0::1::INSTR')
    instrumento.timeout = 5000
    print('Conectado com sucesso')
except pyvisa.VisaIOError:
    print("Falha ao conectar com sucesso")
    exit()

ident = instrumento.query('*IDN?')
print(ident)

instrumento.write('*RST')

instrumento.write('CURR 1')
set_curr = instrumento.query('CURR?')
instrumento.write('OUTP ON')
time.sleep(2)

measured_curr = instrumento.query('CURR?')
print(f'Corrente medida: {measured_curr}')
print(f'Corrente configurada: {set_curr}')
instrumento.write('OUTP OFF')
instrumento.close()
rm.close()
